cmake_minimum_required(VERSION 3.16)
project(betterscale_prefill_microflow LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(SOC_VERSION Ascend910B2)
set(ASCEND_CANN_PACKAGE_PATH "" CACHE PATH "Explicit qualified CANN toolkit")
if(NOT ASCEND_CANN_PACKAGE_PATH)
  message(FATAL_ERROR "Set ASCEND_CANN_PACKAGE_PATH to the qualified CANN toolkit")
endif()
if(NOT EXISTS "${CATLASS_ROOT}/include/catlass/catlass.hpp")
  message(FATAL_ERROR "Set CATLASS_ROOT to the pinned donor CATLASS checkout")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(bs_microflow_kkt SHARED kkt_varlen.cpp)
ascendc_library(bs_microflow_wu SHARED wu2.cpp)
foreach(target bs_microflow_kkt bs_microflow_wu)
  ascendc_include_directories(${target} PRIVATE ${CATLASS_ROOT}/include)
  ascendc_compile_options(${target} PRIVATE -O2 --cce-auto-sync=off -Wno-deprecated-declarations)
endforeach()
